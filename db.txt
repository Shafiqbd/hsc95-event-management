----------------------------------------------------------------------
-- User Creation Function
----------------------------------------------------------------------

CREATE OR REPLACE FUNCTION hsc95.sp_insert_user(
    p_fullname character varying(200),
    p_loginname character varying(100),
    p_password character varying(128),
    p_email character varying(250),
    p_contactnumber character varying(11),
    p_usertype numeric(1,0),
    p_userstatus numeric(1,0) DEFAULT 1
)
RETURNS integer AS $$
DECLARE
    v_userid integer;
    v_hashed_password text;
BEGIN
    -- 1. Encrypt the password using Blowfish (bf) algorithm with an auto-generated salt
    v_hashed_password := crypt(p_password, gen_salt('bf', 8));

    -- 2. Insert the record into your T_Users table
    INSERT INTO hsc95."T_Users" (
        "FullName", 
        "LoginName", 
        "Password", 
        "Email", 
        "ContactNumber", 
        "UserType", 
        "UserStatus"
    )
    VALUES (
        p_fullname, 
        p_loginname, 
        v_hashed_password, 
        p_email, 
        p_contactnumber, 
        p_usertype, 
        p_userstatus
    )
    RETURNING "UserId" INTO v_userid;

    -- 3. Return the newly generated UserId
    RETURN v_userid;
END;
$$ LANGUAGE plpgsql;


----------------------------------------------------------------------
-- User List Function
----------------------------------------------------------------------

CREATE OR REPLACE FUNCTION hsc95.sp_get_user_list()
RETURNS TABLE (
    "UserId" integer,
    "FullName" character varying(200),
    "LoginName" character varying(100),
    "UserTypeName" character varying(15)
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        u."UserId",
        u."FullName",
        u."LoginName",
        t."UserTypeName"
    FROM hsc95."T_Users" u
    INNER JOIN hsc95."T_UserType" t ON u."UserType" = t."UserTypeId"
    ORDER BY u."UserId" ASC;
END;
$$ LANGUAGE plpgsql;

----------------------------------------------------------------------
-- User Login Function
----------------------------------------------------------------------

CREATE OR REPLACE FUNCTION hsc95.sp_user_login(
    p_loginname character varying(100),
    p_password character varying(128)
)
RETURNS TABLE (
    "UserId" integer,
    "FullName" character varying(200),
    "LoginName" character varying(100),
    "ContactNumber" numeric(11,0),
    "EmailAddress" character varying(250),
    "UserTypeName" character varying(15)
) AS $$
DECLARE
    v_userid integer;
    v_loginname character varying(100);
BEGIN
    -- 1. Check if the user exists and credentials match, save identifier details
    SELECT u."UserId", u."LoginName"
    INTO v_userid, v_loginname
    FROM hsc95."T_Users" u
    WHERE u."LoginName" = p_loginname
      AND u."Password" = crypt(p_password, u."Password");

    -- 2. If a matching row was found, insert the audit log first
    IF FOUND THEN
        INSERT INTO hsc95."T_Events" ("EventType", "EventText")
        VALUES (
            'Login', 
            'User ' || v_loginname || ' (ID: ' || v_userid || ') successfully logged in.'
        );
    END IF;

    -- 3. Return the exact matching rows dynamically via RETURN QUERY
    RETURN QUERY
    SELECT 
        u."UserId",
        u."FullName",
        u."LoginName",
        u."ContactNumber",
        u."Email" AS "EmailAddress",
        t."UserTypeName"
    FROM hsc95."T_Users" u
    INNER JOIN hsc95."T_UserType" t ON u."UserType" = t."UserTypeId"
    WHERE u."UserId" = v_userid; -- Reuses the validated UserId
    
END;
$$ LANGUAGE plpgsql;

----------------------------------------------------------------------
-- Form Photo Entry Function
----------------------------------------------------------------------

CREATE OR REPLACE FUNCTION hsc95.sp_insert_event_form_photo(
    p_entryid numeric(20,0),
    p_photopath text,
    p_photostatus numeric(1,0) DEFAULT 1
)
RETURNS bigint AS $$
DECLARE
    v_new_photoid bigint;
BEGIN
    -- 1. If the photo status to be inserted is active (1), 
    --    deactivate all previous photos for this specific EntryId
    IF p_photostatus = 1 THEN
        UPDATE hsc95."T_Photos"
        SET "PhotoStatus" = 0
        WHERE "EntryId" = p_entryid;
    END IF;

    -- 2. Generate a new incremental PhotoId since no auto-sequence is configured
    SELECT COALESCE(MAX("PhotoId"), 0) + 1
    INTO v_new_photoid
    FROM hsc95."T_Photos";

    -- 3. Insert the new photo record
    INSERT INTO hsc95."T_Photos" (
        "PhotoId",
        "EntryId",
        "PhotoPath",
        "PhotoStatus"
    )
    VALUES (
        v_new_photoid,
        p_entryid,
        p_photopath,
        p_photostatus
    );

    -- Return the newly generated PhotoId
    RETURN v_new_photoid;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION hsc95.sp_prepare_sms()
RETURNS TABLE (
    "SMS" text,
    "MobileNo" character varying(11)
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        'Dear ' || "MemberName" || ', Your registration is complete. Your token no will be send very soon. Thank you for participate. HSC 95 Society (30 Years celebration)'::text AS "SMS",
        "MobileNo"
    FROM hsc95."T_EventForm"
    ORDER BY "EntryId" ASC;
END;
$$ LANGUAGE plpgsql;

----------------------------------------------------------------------
-- Entry Pass Cretion (JPG Image) Function
----------------------------------------------------------------------

CREATE OR REPLACE FUNCTION hsc95.sp_prepare_entrypass()
RETURNS TABLE (
    "MemberName" text,
    "FormNo" numeric(5,0),
    "MobileNo" character varying(11),
    "PhotoPath" text
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        f."MemberName",
        f."FormNo",
        f."MobileNo",
        p."PhotoPath"
    FROM hsc95."T_EventForm" f
    -- Join numeric(20,0) with integer; PostgreSQL automatically handles this numeric comparison safely
    INNER JOIN hsc95."T_Photos" p ON f."EntryId" = p."EntryId"
    -- Only return the active profile image
    WHERE p."PhotoStatus" = 1
    ORDER BY f."EntryId" ASC;
END;
$$ LANGUAGE plpgsql;
